import os
from flask import Flask, request, jsonify
import pymysql.cursors
import pymysql

app = Flask(__name__)

app.config['MAX_CONTENT_LENGTH'] = 16 * 1024 * 1024
app.config['MAX_FILE_PART_HANDLES'] = 100
app.config['MAX_FORM_PART_HEADERS'] = 1000

MYSQL_HOST = os.getenv('MYSQL_HOST', 'localhost')
MYSQL_USER = os.getenv('MYSQL_USER', 'root')
MYSQL_PASSWORD = os.getenv('MYSQL_PASSWORD', 'password')
MYSQL_DB = os.getenv('MYSQL_DB', 'mydb')
MYSQL_PORT = int(os.getenv('MYSQL_PORT', 3306))

# Ajout de variables d'environnement pour la configuration SSL/TLS de la base de données
# Ces chemins devraient pointer vers des fichiers de certificat et clé sur le système de fichiers du serveur
MYSQL_SSL_CA = os.getenv('MYSQL_SSL_CA')      # Chemin vers le fichier du certificat de l'autorité de certification
MYSQL_SSL_CERT = os.getenv('MYSQL_SSL_CERT')  # Chemin vers le fichier du certificat client (si nécessaire)
MYSQL_SSL_KEY = os.getenv('MYSQL_SSL_KEY')    # Chemin vers le fichier de la clé privée client (si nécessaire)

def get_db_connection():
    try:
        connection_params = {
            'host': MYSQL_HOST,
            'user': MYSQL_USER,
            'password': MYSQL_PASSWORD,
            'database': MYSQL_DB,
            'port': MYSQL_PORT,
            'cursorclass': pymysql.cursors.DictCursor
        }
        
        # Activer SSL/TLS pour la connexion à la base de données si un certificat CA est fourni.
        # C'est une bonne pratique de sécurité pour chiffrer les communications et vérifier l'identité du serveur.
        if MYSQL_SSL_CA:
            connection_params['ssl_ca'] = MYSQL_SSL_CA
            if MYSQL_SSL_CERT:
                connection_params['ssl_cert'] = MYSQL_SSL_CERT
            if MYSQL_SSL_KEY:
                connection_params['ssl_key'] = MYSQL_SSL_KEY

        connection = pymysql.connect(**connection_params)
        return connection
    except Exception as e:
        app.logger.error(f"Error connecting to database: {e}")
        return None

@app.route('/user_data', methods=['GET'])
def get_user_data():
    user_id = request.args.get('user_id', type=int)

    if user_id is None:
        return jsonify({"error": "user_id is required"}), 400

    connection = get_db_connection()
    if connection is None:
        return jsonify({"error": "Database connection failed"}), 500

    try:
        with connection.cursor() as cursor:
            # Les requêtes paramétrées (avec %s) protègent contre les injections SQL
            sql = "SELECT id, username, email FROM users WHERE id = %s"
            cursor.execute(sql, (user_id,))
            result = cursor.fetchone()

            if result:
                return jsonify(result), 200
            else:
                return jsonify({"message": "User not found"}), 404
    except pymysql.Error as e:
        app.logger.error(f"Database error: {e}")
        return jsonify({"error": "An internal database error occurred"}), 500
    finally:
        if connection:
            connection.close()

@app.route('/create_user', methods=['POST'])
def create_user():
    data = request.get_json()
    if not data:
        return jsonify({"error": "Invalid JSON input"}), 400

    username = data.get('username')
    email = data.get('email')
    password_hash = data.get('password_hash')

    if not all([username, email, password_hash]):
        return jsonify({"error": "Missing username, email, or password_hash"}), 400

    connection = get_db_connection()
    if connection is None:
        return jsonify({"error": "Database connection failed"}), 500

    try:
        with connection.cursor() as cursor:
            # Les requêtes paramétrées (avec %s) protègent contre les injections SQL
            sql = "INSERT INTO users (username, email, password_hash) VALUES (%s, %s, %s)"
            cursor.execute(sql, (username, email, password_hash))
            connection.commit()
            return jsonify({"message": "User created successfully", "id": cursor.lastrowid}), 201
    except pymysql.IntegrityError as e:
        app.logger.error(f"Integrity error: {e}")
        return jsonify({"error": "User with this email or username already exists"}), 409
    except pymysql.Error as e:
        app.logger.error(f"Database error: {e}")
        return jsonify({"error": "An internal database error occurred"}), 500
    finally:
        if connection:
            connection.close()

@app.route('/')
def health_check():
    return "Application is running!"
